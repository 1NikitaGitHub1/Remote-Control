class ControlButton(Button): 
    """ Kivy's Button with functions 
    1. keyboard processing - Nope only bottom key and without hold 
    """ 
     
    def __init__(self, action : str, **kwargs) -> None: 
        super().__init__(**kwargs) 
        self.action = action 
        self._keyboard = Window.request_keyboard(self._keyboard_closed, self) 
        self._keyboard.bind(on_key_down=self.press) 
 
 
    def _keyboard_closed(self) -> None: 
        self._keyboard.unbind(on_key_down=self.press) 
        self._keyboard = None 
 
 
    def press(self, keyboard, keycode : chr, text, modifiers) -> None: 
        # Handle key presses for 'W', 'A', 'S', 'D' 
        print(self.action) 
        if keycode[1] == 'd' and self.action == 'right': 
            print("right") 
            self.trigger_action() 
        elif keycode[1] == 'a' and self.action == 'left': 
            print("left") 
            self.trigger_action() 
        elif keycode[1] == 'w' and self.action == 'up': 
            print("up") 
            self.trigger_action() 
        elif keycode[1] == 's' and self.action == 'bottom': 
            print("bottom") 
            self.trigger_action() 
        return True
